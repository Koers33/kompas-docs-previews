        uuid id PK
        uuid counterpartyId FK
        string agreementType "ISDA | CSA | GMRA | GMSLA | CREDIT_AGREEMENT | CLIENT_MANDATE"
        string status "NEGOTIATION | ACTIVE | SUSPENDED | TERMINATED"
        date signedDate
        date effectiveDate
        date expiryDate
        string governingLaw
        uuid evidencePackId FK
    }
    LegalClause {
        uuid id PK
        uuid legalAgreementId FK
        string clauseType "NETTING | COLLATERAL | CLOSE_OUT | COVENANT | REPRESENTATION"
        string clauseContent
        boolean isOperationallyEnforced
    }
    Covenant {
        uuid id PK
        uuid legalAgreementId FK
        string covenantType "FINANCIAL | OPERATIONAL | REGULATORY | COMPLIANCE"
        decimal thresholdValue
        string metricDefinition
        string status "COMPLIANT | BREACHED | WAIVED"
    }
    LegalObligation {
        uuid id PK
        uuid legalAgreementId FK
        string obligationType "PAYMENT | SETTLEMENT | REPORTING | COLLATERAL | NOTIFICATION"
        date dueDate
        string status "PENDING | FULFILLED | OVERDUE | BREACHED"
    }
    CreditEventRecord {
        uuid id PK
        uuid counterpartyId FK
        string creditEventType "FAILURE_TO_PAY | BANKRUPTCY | RESTRUCTURING | RATING_DOWNGRADE"
        date eventDate
        date detectedAt
        string status "DETECTED | FROZEN | CLOSE_OUT_INITIATED | CLOSE_OUT_COMPLETED"
        uuid legalNoticeId FK
    }
    CloseOutNotice {
        uuid id PK
        uuid creditEventRecordId FK
        string noticeType "ISDA_DEFAULT_NOTICE | TERMINATION_NOTICE | CLOSE_OUT_NOTICE"
        uuid approvedByLegalId FK
        uuid approvedByManagerId FK
        uuid boardApprovalId FK
        date sentAt
        uuid evidencePackId FK
    }
    DisputeRecord {
        uuid id PK
        uuid relatedEntityId FK
        string disputeType "MARGIN | VALUATION | SETTLEMENT | TRADE | TERMINATION"
        string status "OPEN | NEGOTIATING | ESCALATED | RESOLVED"
        uuid evidencePackId FK
    }
    RepresentationAuthority {
        uuid id PK
        uuid clientId FK
        uuid counterpartyId FK
        string authorityType "TRADING | SETTLEMENT | MARGIN | REPORTING"
        string scope
        date validFrom
        date validTo
        boolean isActive
    }
